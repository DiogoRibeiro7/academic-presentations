\begin{itemize}
\item Transformed statistics from analytical to computational
\item Enabled Bayesian inference for complex models
\item Made high-dimensional problems tractable
\item Foundation of modern ML/AI
\end{itemize}
\end{block}
\end{column}

\end{alertblock}

\begin{frame}{The Fundamental Sampling Challenge}
  \begin{itemize}
    \item Bayesian and latent-variable models often require expectations with respect to a target density known only up to a normalizing constant.
    \item Direct, rejection, importance, and deterministic quadrature methods can all be excellent in suitable problems; their performance deteriorates for different reasons as dimension, tail behavior, multimodality, or mismatch increase.
    \item MCMC replaces independent sampling by a dependent stochastic process designed to have the target distribution as invariant distribution.
    \item This does not make difficult targets automatically easy: poor parameterization, separated modes, heavy tails, funnels, and strong correlations can still produce severe computational problems.
  \end{itemize}

  \begin{block}{What MCMC buys us}
    Under appropriate ergodicity conditions, averages along the chain can estimate posterior expectations without evaluating the normalizing constant.
  \end{block}
\end{frame}

\section{Mathematical Foundations}


\begin{frame}{Markov Chain Theory}
\begin{definition}[Markov Chain]
A sequence $\{X_n\}_{n \geq 0}$ is a Markov chain if:
\[P(X_{n+1} = x_{n+1} | X_0, \ldots, X_n) = P(X_{n+1} = x_{n+1} | X_n)\]
\end{definition}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Key Concepts:}
\begin{itemize}
\item \textbf{Transition kernel:} $P(x, A) = P(X_{n+1} \in A | X_n = x)$
\item \textbf{Chapman-Kolmogorov:} $P^n(x,A) = \int P^{n-1}(x,dy) P(y,A)$
\item \textbf{Invariant distribution:} $\pi(A) = \int \pi(dx) P(x,A)$
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\begin{theorem}[Ergodic Theorem]
If the chain is irreducible, aperiodic, and positive recurrent, then:
\[\lim_{n \to \infty} \frac{1}{n} \sum_{i=1}^n f(X_i) = \int f(x) \pi(dx) \quad \text{a.s.}\]
\end{theorem}
\end{column}
\end{columns}

\vspace{0.3cm}
\begin{block}{Detailed Balance Condition}
\[\pi(x) P(x, dy) = \pi(y) P(y, dx)\]
Sufficient (but not necessary) for $\pi$ to be stationary.
\end{block}
\end{frame}

\begin{frame}{Convergence Theory and Rates}
  \begin{itemize}
    \item Invariance of $\pi$ is necessary but not sufficient for reliable Monte Carlo estimation; the chain must also explore the relevant state space appropriately.
    \item Irreducibility, recurrence, aperiodicity, drift/minorization conditions, and geometric ergodicity are examples of conditions used in different convergence results.
    \item For finite reversible chains, spectral gaps characterize important convergence properties. General state-space chains require more careful operator or drift-based arguments.
    \item A Markov-chain central limit theorem for an estimand $f$ requires conditions beyond stationarity:
      \[\sqrt{n}(\bar f_n-\pi(f)) \Rightarrow \mathcal{N}(0,\sigma_f^2),\]
      where $\sigma_f^2$ includes the autocovariance contribution of the chain.
  \end{itemize}

  \begin{alertblock}{Practical implication}
    Convergence is estimand- and algorithm-dependent. A chain can explore some posterior summaries adequately while failing badly for tails, rare modes, or strongly correlated directions.
  \end{alertblock}
\end{frame}

\section{Classical Algorithms}


\begin{frame}{Metropolis-Hastings Algorithm}
\begin{algorithm}[H]
\caption{Metropolis-Hastings}
\begin{algorithmic}[1]
\STATE Initialize $x^{(0)}$
\FOR{$t = 0, 1, 2, \ldots$}
\STATE Propose $y \sim q(y | x^{(t)})$
\STATE Compute acceptance probability:
\[
\alpha(x^{(t)}, y) = \min\left(1, \frac{\pi(y) q(x^{(t)} | y)}{\pi(x^{(t)}) q(y | x^{(t)})}\right)
\]
\STATE Accept $x^{(t+1)} = y$ with probability $\alpha$, otherwise $x^{(t+1)} = x^{(t)}$
\ENDFOR
\end{algorithmic}
\end{algorithm}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Key Variants:}
\begin{itemize}
\item \textbf{Random Walk:} $q(y|x) = q(y-x)$
\item \textbf{Independence:} $q(y|x) = q(y)$
\item \textbf{Langevin:} $q(y|x) \propto \exp(-\|y-x-\frac{\epsilon^2}{2}\nabla\log\pi(x)\|^2/(2\epsilon^2))$
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\begin{block}{Asymptotic Scaling Insight}
For random-walk Metropolis under specific high-dimensional product-target assumptions, asymptotic analyses yield the well-known acceptance rate near $0.234$ and proposal scale of order $d^{-1/2}$. These are model-specific theoretical results, not universal tuning targets for arbitrary MCMC algorithms or posteriors.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Advanced Proposal Strategies}
\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Adaptive Metropolis:}
\[C_n = \frac{s_d}{n} \sum_{i=1}^n (X_i - \bar{X}_n)(X_i - \bar{X}_n)^T + s_d \epsilon_n I_d\]

where $s_d = (2.38)^2/d$ and $\epsilon_n \to 0$.

\vspace{0.3cm}
\textbf{Advantages:}
\begin{itemize}
\item Automatic tuning
\item Adapts to target geometry
\item Maintains ergodicity
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Multiple-try Metropolis:}
\begin{enumerate}
\item Generate $k$ proposals: $y_1, \ldots, y_k \sim q(\cdot | x)$
\item Select $y_j$ with probability $\propto \pi(y_j)$
\item Generate reference set from $y_j$
\item Accept/reject based on ratio of weights
\end{enumerate}

\vspace{0.3cm}
\begin{block}{Benefits}
\begin{itemize}
\item Higher acceptance rates
\item Better exploration
\item Parallelizable proposals
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Gibbs Sampling and Blocking}
\textbf{Standard Gibbs Sampling:}
\begin{align}
X_1^{(t+1)} &\sim \pi(x_1 | X_2^{(t)}, X_3^{(t)}, \ldots, X_d^{(t)}) \\
X_2^{(t+1)} &\sim \pi(x_2 | X_1^{(t+1)}, X_3^{(t)}, \ldots, X_d^{(t)}) \\
&\vdots \\
X_d^{(t+1)} &\sim \pi(x_d | X_1^{(t+1)}, X_2^{(t+1)}, \ldots, X_{d-1}^{(t+1)})
\end{align}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Blocking Strategies:}
\begin{itemize}
\item \textbf{Random scan:} Update components randomly
\item \textbf{Block Gibbs:} Update correlated components together
\item \textbf{Collapsed Gibbs:} Integrate out auxiliary variables
\end{itemize}

\textbf{Example: Hierarchical Models}
\[\theta_i | \mu, \tau \sim N(\mu, \tau^2)\]
Block $(\mu, \tau)$ for better mixing.
\end{column}
\begin{column}{0.5\textwidth}
\begin{alertblock}{Performance Considerations}
\begin{itemize}
\item \textcolor{crimson}{Slow mixing:} High posterior correlations
\item \textcolor{forest}{Fast mixing:} Near-independence
\item \textcolor{crimson}{Curse:} $O(d^2)$ scaling with correlation
\end{itemize}
\end{alertblock}

\textbf{Acceleration Techniques:}
\begin{itemize}
\item Parameter expansion
\item Reparameterization
\item Auxiliary variable methods
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\section{Modern Algorithms}

\begin{frame}{Hamiltonian Monte Carlo}
\textbf{Physical Intuition:} Frictionless particle on curved surface

\vspace{0.3cm}
\begin{columns}
\begin{column}{0.6\textwidth}
\textbf{Hamiltonian System:}
\begin{align}
H(q, p) &= U(q) + K(p) \\
U(q) &= -\log \pi(q) \text{ (potential energy)} \\
K(p) &= \frac{1}{2} p^T M^{-1} p \text{ (kinetic energy)}
\end{align}

\textbf{Hamilton's equations:}
\begin{align}
\frac{dq}{dt} &= \frac{\partial H}{\partial p} = M^{-1} p \\
\frac{dp}{dt} &= -\frac{\partial H}{\partial q} = -\nabla U(q)
\end{align}
\end{column}
\begin{column}{0.4\textwidth}
\begin{block}{Leapfrog Integrator}
\begin{align}
p_{t+\epsilon/2} &= p_t - \frac{\epsilon}{2} \nabla U(q_t) \\
q_{t+\epsilon} &= q_t + \epsilon M^{-1} p_{t+\epsilon/2} \\
p_{t+\epsilon} &= p_{t+\epsilon/2} - \frac{\epsilon}{2} \nabla U(q_{t+\epsilon})
\end{align}
\end{block}
\end{column}
\end{columns}

\vspace{0.3cm}
\begin{alertblock}{Key Advantages}
\begin{itemize}
\item \textbf{Distant proposals:} Can traverse typical set efficiently
\item \textbf{Energy-informed proposals:} Numerical integration can support long-distance moves with controlled Metropolis correction
\item \textbf{Gradient information:} Exploits structure of target distribution
\end{itemize}
\end{alertblock}
\end{frame}

\begin{frame}{No-U-Turn Sampler (NUTS)}
\textbf{Problem with HMC:} Manual tuning of step size $\epsilon$ and number of steps $L$

\begin{algorithm}[H]
\caption{NUTS Algorithm (Simplified)}
\begin{algorithmic}[1]
\STATE Sample momentum $p_0 \sim N(0, M)$
\STATE Set $q_- = q_+ = q_0$, $p_- = p_+ = p_0$
\STATE Build trajectory by doubling until U-turn criterion:
\WHILE{no U-turn detected}
\STATE Double trajectory length in random direction
\STATE Check stopping criterion: $(q_+ - q_-) \cdot p_+ < 0$ or $(q_+ - q_-) \cdot p_- < 0$
\ENDWHILE
\STATE Sample uniformly from valid points in trajectory
\end{algorithmic}
\end{algorithm}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Automatic Adaptation:}
\begin{itemize}
\item \textbf{Step size:} Dual averaging to target acceptance rate
\item \textbf{Mass matrix:} Regularized sample covariance
\item \textbf{Trajectory length:} Dynamic stopping criterion
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\begin{block}{Performance Benefits}
\begin{itemize}
\item Adapts step size and mass-matrix information during warm-up in common implementations
\item Chooses path length dynamically rather than fixing a leapfrog count in advance
\item Often effective for differentiable continuous posteriors with suitable parameterization
\item Still requires diagnostics; divergences and pathological geometry are not removed by automation
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Diagnostics and Convergence}

\begin{frame}{Convergence Assessment: Multiple Diagnostics, One Estimand}
  \begin{itemize}
    \item Use multiple chains from dispersed initial states and inspect rank-normalized split-$\hat R$; values materially above 1 indicate between-chain disagreement or nonstationarity.
    \item Bulk and tail effective sample sizes quantify different aspects of Monte Carlo efficiency. Required ESS depends on the posterior summary and precision needed for the analysis.
    \item Monte Carlo standard error should be small relative to the inferential scale of interest; this is more meaningful than a universal sample-count threshold.
    \item For HMC/NUTS, inspect divergent transitions, tree-depth saturation, energy diagnostics, and problematic parameter correlations.
    \item Rank plots, trace behavior, and posterior predictive checks answer complementary questions; none is a proof that the posterior model itself is correct.
  \end{itemize}

  \begin{alertblock}{Diagnostic thresholds}
    A rank-normalized split-$\hat R$ near 1 (often using 1.01 as a warning threshold) is useful, but passing a threshold does not certify convergence. Diagnostics should be interpreted jointly and relative to the estimand.
  \end{alertblock}
\end{frame}

class AdaptiveMetropolis:
    def __init__(self, target_log_prob, dim):
        self.target_log_prob = target_log_prob
        self.dim = dim
        self.cov = np.eye(dim) * 0.1
        self.mean = np.zeros(dim)
        
    def sample(self, n_samples, x0):
        samples = np.zeros((n_samples, self.dim))
        samples[0] = x0
        current_x = x0
        current_logp = self.target_log_prob(x0)
        
        accepted = 0
        for i in range(1, n_samples):
            # Propose
            proposal = np.random.multivariate_normal(
                current_x, self.cov)
            proposal_logp = self.target_log_prob(proposal)
            
            # Accept/reject
            if (np.log(np.random.random()) < 
                proposal_logp - current_logp):
                current_x = proposal
                current_logp = proposal_logp
                accepted += 1
                
            samples[i] = current_x
            
            # Adapt covariance
            if i % 100 == 0 and i > 200:
                self.adapt_covariance(samples[:i])
                
        return samples, accepted / n_samples
\end{lstlisting}

\section{Conclusions}

\begin{frame}{Synthesis}
  \begin{itemize}
    \item MCMC correctness has two layers: the transition kernel must target the desired distribution, and the realized computation must explore that distribution well enough for the estimand of interest.
    \item Metropolis--Hastings, Gibbs, HMC, NUTS, tempering, and specialized samplers exploit different structure; no sampler dominates across posterior geometries.
    \item Parameterization can matter as much as the nominal algorithm. Centered versus non-centered forms, scaling, constraints, and posterior geometry can determine practical efficiency.
    \item $\hat R$, ESS, MCSE, rank plots, divergences, and predictive checks are complementary diagnostics. Passing a numerical threshold is not a proof of convergence or model adequacy.
    \item Monte Carlo precision should be chosen relative to the inferential target: additional computation is useful only when simulation error is material compared with posterior uncertainty or decision sensitivity.
  \end{itemize}
\end{frame}
