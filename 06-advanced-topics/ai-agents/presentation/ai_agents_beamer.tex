\section{What an AI Agent Is}
% ================================================================

\begin{frame}{Agent vs. Single LLM Call}
  \textbf{Single LLM call:}
  \begin{itemize}
    \item One prompt in, one response out
    \item No persistent state
    \item No autonomous tool use
  \end{itemize}

  \vspace{0.7em}

  \textbf{AI agent:}
  \begin{itemize}
    \item Goal-directed loop: plan, act, observe, adapt
    \item Uses tools (search, code, APIs, databases)
    \item Maintains memory and context over time
    \item Handles failures and retries
  \end{itemize}
\end{frame}

\begin{frame}{Core Agent Loop}
  \begin{definitionbox}{Agent Control Loop}
    Repeat until done:
    \begin{enumerate}
      \item Interpret user goal and current state
      \item Decide next action
      \item Execute tool or model call
      \item Evaluate outcome against success criteria
      \item Update memory and continue or stop
    \end{enumerate}
  \end{definitionbox}

  \vspace{0.6em}
  \begin{block}{Design implication}
    The loop logic is software engineering. The LLM is one component, not the whole system.
  \end{block}
\end{frame}

\section{System Architecture}
% ================================================================

\begin{frame}{Reference Architecture}
  \begin{itemize}
    \item \textbf{Orchestrator:} state machine and policy logic
    \item \textbf{Model layer:} LLM(s) for reasoning and generation
    \item \textbf{Tool layer:} deterministic actions (APIs, DB, shell, workflows)
    \item \textbf{Memory layer:} short-term context and long-term retrieval
    \item \textbf{Safety layer:} permissions, validation, guardrails
    \item \textbf{Observability layer:} logs, traces, quality metrics, cost
  \end{itemize}
\end{frame}

\begin{frame}{Choose the Right Agent Pattern}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Simple Router}
    \begin{itemize}
      \item Fast, low complexity
      \item Great for FAQ and triage
    \end{itemize}

    \textbf{Tool-Using ReAct Agent}
    \begin{itemize}
      \item Think-act-observe loops
      \item Works for research and operations
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Planner + Executor}
    \begin{itemize}
      \item Better for long tasks
      \item Easier to checkpoint
    \end{itemize}

    \textbf{Multi-Agent Workflow}
    \begin{itemize}
      \item Specialized roles
      \item Higher coordination overhead
    \end{itemize}
  \end{columns}
\end{frame}

\section{Building Blocks}
% ================================================================

\begin{frame}[fragile]{Tool Interface Contract}
  \textbf{Keep tool calls structured and typed.}

  \begin{lstlisting}[language=Python, caption=Minimal tool schema]
{
  "name": "get_customer_invoice",
  "description": "Fetch invoice by id",
  "input_schema": {
    "type": "object",
    "properties": {
      "invoice_id": {"type": "string"}
    },
    "required": ["invoice_id"]
  }
}
  \end{lstlisting}

  \begin{itemize}
    \item Validate input before execution
    \item Return explicit success/error payloads
    \item Make tools idempotent when possible
  \end{itemize}
\end{frame}

\begin{frame}{Memory Strategy}
  \textbf{Use layered memory, not one giant prompt.}
  \begin{itemize}
    \item \textbf{Working memory:} current task context
    \item \textbf{Session memory:} recent interactions and outcomes
    \item \textbf{Long-term memory:} indexed facts, policies, and user preferences
  \end{itemize}

  \vspace{0.6em}
  \begin{alertbox}{Common failure mode}
    Storing everything in conversation history causes context drift, latency, and hallucinations.
  \end{alertbox}
\end{frame}

\begin{frame}{Planning and Decomposition}
  \begin{enumerate}
    \item Convert vague intent into explicit success criteria
    \item Split task into atomic steps with clear dependencies
    \item Execute smallest safe next action
    \item Re-plan when observations diverge from expectations
  \end{enumerate}

  \vspace{0.6em}
  \begin{examplebox}{Example}
    User goal: "Prepare Q2 churn analysis"

    Agent plan: fetch data $\rightarrow$ validate schema $\rightarrow$ run analysis $\rightarrow$ generate slides $\rightarrow$ request approval.
  \end{examplebox}
\end{frame}

\section{Reliability and Safety}
% ================================================================

\begin{frame}{Guardrails You Actually Need}
  \begin{itemize}
    \item \textbf{Permission boundaries:} role-based tool access
    \item \textbf{Input/output validation:} schema + policy checks
    \item \textbf{Human-in-the-loop gates:} approvals for destructive actions
    \item \textbf{Timeouts and retries:} bounded recovery behavior
    \item \textbf{Fallback paths:} deterministic flows when agent confidence is low
  \end{itemize}
\end{frame}

\begin{frame}{Evaluation and Monitoring}
  \textbf{Evaluate offline and in production.}

  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Quality metrics}
    \begin{itemize}
      \item Task success rate
      \item Tool-call precision
      \item Hallucination rate
      \item Recovery rate after failure
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Operational metrics}
    \begin{itemize}
      \item Latency per step
      \item Cost per task
      \item Token consumption
      \item Escalation frequency
    \end{itemize}
  \end{columns}
\end{frame}

\section{From Prototype to Production}
% ================================================================

\begin{frame}{Implementation Roadmap}
  \begin{enumerate}
    \item Start with one narrow, high-value workflow
    \item Implement deterministic tools first
    \item Add a constrained agent loop with strict schemas
    \item Instrument traces and evaluate against test tasks
    \item Add memory and planner only when baseline is stable
    \item Expand scope after reliability targets are met
  \end{enumerate}
\end{frame}

\begin{frame}{Common Anti-Patterns}
  \begin{itemize}
    \item Building a general autonomous agent too early
    \item Missing tool contracts and error taxonomies
    \item No golden test set for regressions
    \item Prompt-only fixes for architectural problems
    \item Ignoring cost controls until deployment
  \end{itemize}

  \vspace{0.6em}
  \begin{alertbox}{Pragmatic rule}
    If behavior must be guaranteed, move it into deterministic code, not prompt text.
  \end{alertbox}
\end{frame}

\begin{frame}{Closing}
  \begin{block}{Key message}
    A production AI agent is a software system with an LLM component, not an LLM prompt with extra steps.
  \end{block}

  \vspace{0.8em}
  \textbf{Checklist:}
  \begin{itemize}
    \item Clear task boundaries
    \item Typed tool interfaces
    \item Explicit state and memory strategy
    \item Guardrails and approval gates
    \item Continuous evaluation and monitoring
  \end{itemize}
\end{frame}
