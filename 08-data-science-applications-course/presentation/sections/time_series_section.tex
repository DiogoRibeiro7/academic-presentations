\begin{frame}{What Makes Time Series Different?}
A time series is an ordered sequence $\{y_t\}_{t=1}^T$ observed over time.

\begin{itemize}
  \item ordering determines which information was available at each decision or forecast origin;
  \item observations may be serially dependent, seasonally structured, or subject to evolving regimes;
  \item random train/test shuffling is inappropriate when it leaks future information into training;
  \item not every time series is strongly autocorrelated, and not every cross-sectional dataset is independent.
\end{itemize}

\begin{alertblock}{Validation principle}
Reproduce the deployment information set: preserve time ordering, forecast horizon, feature availability, and retraining schedule.
\end{alertblock}
\end{frame}

\begin{frame}{Common Time-Series Structures}
\begin{itemize}
  \item \textbf{Univariate}: one measured process over time.
  \item \textbf{Multivariate}: several jointly observed variables over time.
  \item \textbf{Panel / multiple series}: many entities each observed repeatedly.
  \item \textbf{Irregular or event time}: observations arrive at non-uniform timestamps.
\end{itemize}

\textbf{Examples}
\begin{itemize}
  \item hourly temperature or heart-rate traces;
  \item monthly product demand;
  \item web traffic at minute resolution;
  \item many stores, assets, or patients followed over time.
\end{itemize}
\end{frame}

\begin{frame}{Time Series Components}
\textbf{Decomposition is a descriptive modeling device, not a universal data-generating law.}

\textbf{Additive form:}
\[
  y_t = T_t + S_t + C_t + I_t.
\]

\begin{itemize}
  \item $T_t$: long-run trend or smooth level;
  \item $S_t$: recurring seasonal structure;
  \item $C_t$: longer, non-periodic cycles;
  \item $I_t$: remainder not explained by the chosen decomposition.
\end{itemize}

\textbf{Multiplicative form:}
\[
  y_t = T_t \times S_t \times C_t \times I_t,
\]
which can be useful when seasonal amplitude scales with the series level.
\end{frame}

\begin{frame}{Stationarity: The Foundation}
\textbf{Why stationarity matters for time series modeling}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Definition:}

A time series is \textbf{strictly stationary} if:
\[
P(y_{t_1}, \ldots, y_{t_k}) = P(y_{t_1+h}, \ldots, y_{t_k+h})
\]
for all $t_1, \ldots, t_k$ and $h$.

\vspace{0.2cm}

\textbf{Weak (Covariance) Stationarity:}
\begin{enumerate}
\item $\E[y_t] = \mu$ (constant mean)
\item $\Var[y_t] = \sigma^2$ (constant variance)
\item $\Cov[y_t, y_{t-k}] = \gamma_k$ (depends only on lag $k$)
\end{enumerate}

\vspace{0.3cm}

\textbf{Why It Matters:}
\begin{itemize}
\item Most models assume stationarity
\item Enables statistical inference
\item Allows generalization to future
\item Non-stationary $\to$ spurious results
\end{itemize}
\end{column}

\begin{column}{0.5\textwidth}
\textbf{Testing for Stationarity:}

\begin{block}{Augmented Dickey-Fuller (ADF) Test}
Tests null hypothesis of unit root:
\[
H_0: \text{Non-stationary (unit root)}
\]
\[
H_1: \text{Stationary}
\]

Reject $H_0$ if p-value $< 0.05$
\end{block}

\vspace{0.2cm}

\textbf{Making Series Stationary:}

\begin{enumerate}
\item \textcolor{forest}{\textbf{Differencing:}}
   \[
   \Delta y_t = y_t - y_{t-1}
   \]
   Removes trend

\item \textcolor{navyblue}{\textbf{Log transform:}}
   \[
   \tilde{y}_t = \log(y_t)
   \]
   Stabilizes variance

\item \textcolor{crimson}{\textbf{Seasonal differencing:}}
   \[
   \Delta_s y_t = y_t - y_{t-s}
   \]
   Removes seasonality (period $s$)
\end{enumerate}
\end{column}
\end{columns}

\vspace{0.2cm}

\begin{alertblock}{Key Insight}
Always check stationarity before modeling. Transform if necessary!
\end{alertblock}
\end{frame}

\begin{frame}{ARIMA: Model Structure}
ARIMA$(p,d,q)$ models a differenced series using autoregressive and moving-average terms.

\begin{itemize}
  \item $p$: autoregressive order;
  \item $d$: differencing order;
  \item $q$: moving-average order.
\end{itemize}

\[
  \phi(B)(1-B)^d y_t = \theta(B)\varepsilon_t.
\]

\begin{alertblock}{Differencing}
Differencing is appropriate for certain stochastic trends; it does not mechanically ``make a series stationary'' in every nonstationary setting.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{ARIMA: Identification and Evaluation}
\begin{itemize}
  \item ACF/PACF patterns are useful for proposing candidate orders, not uniquely identifying a finite-sample model.
  \item AIC/BIC compare fitted candidate models under their respective criteria.
  \item Automated order search is a convenience, not a substitute for residual diagnostics and forecast validation.
  \item Seasonal ARIMA adds seasonal AR, differencing, and MA components at period $s$.
\end{itemize}

\begin{lstlisting}[style=code,basicstyle=\ttfamily\small]
from statsmodels.tsa.arima.model import ARIMA

fit = ARIMA(data, order=(1, 1, 1)).fit()
forecast = fit.forecast(steps=12)
\end{lstlisting}

\begin{block}{Evaluation}
Compare candidate models using residual diagnostics and rolling-origin forecast loss at the horizons that matter operationally.
\end{block}
\end{frame}

\begin{frame}{Knowledge Check: Time Series Fundamentals}
\textbf{Multiple Choice Questions}

\vspace{0.3cm}

\textbf{Question 1:} You have daily sales data for a retail store. Which decomposition would you likely use?

\begin{enumerate}[A)]
\item Additive, because sales don't vary
\item \textcolor{forest}{\textbf{Multiplicative, because seasonal variation grows with sales level}}
\item Neither, sales data doesn't have seasonality
\item Use differencing instead
\end{enumerate}

\pause

\begin{block}{Explanation}
\textcolor{forest}{\textbf{Answer: B}} - Retail sales typically show seasonal patterns (holidays, weekends) that grow proportionally with the overall sales level. For example, a 20\% Black Friday boost means \$200 extra when baseline is \$1000, but \$2000 extra when baseline is \$10,000. This multiplicative relationship requires multiplicative decomposition.
\end{block}

\pause

\vspace{0.2cm}

\textbf{Question 2:} ADF test gives p-value = 0.23. What should you do?

\begin{enumerate}[A)]
\item Proceed with modeling, series is stationary
\item \textcolor{forest}{\textbf{Apply differencing to make series stationary}}
\item Use a different test
\item Nothing, p-values don't matter for time series
\end{enumerate}

\pause

\begin{block}{Explanation}
\textcolor{forest}{\textbf{Answer: B}} - p-value = 0.23 > 0.05 means we fail to reject the null hypothesis of non-stationarity. The series likely has a unit root (random walk). Apply differencing ($\Delta y_t = y_t - y_{t-1}$) to remove the trend and achieve stationarity before fitting ARIMA or other models.
\end{block}
\end{frame}

\begin{frame}{Modern Forecasting Model Families}
\begin{itemize}
  \item \textbf{Feature-based ML}: lagged values, rolling summaries, calendar variables, and exogenous covariates fed to tree ensembles or other supervised learners.
  \item \textbf{Structured additive models}: explicit trend, seasonality, holidays, and regressors.
  \item \textbf{State-space models}: latent-state dynamics with recursive filtering and smoothing.
  \item \textbf{Neural sequence models}: RNN/LSTM/GRU, N-BEATS, and Transformer-based architectures.
\end{itemize}

\begin{alertblock}{Model-family choice}
Choose the family around forecast horizon, number of related series, covariate availability, data volume, update frequency, uncertainty requirements, and deployment cost.
\end{alertblock}
\end{frame}

\begin{frame}{Selecting a Forecasting Method}
\begin{table}
\small
\begin{tabular}{p{0.29\textwidth}p{0.63\textwidth}}
\toprule
\textbf{Method family} & \textbf{Useful when} \\
\midrule
ARIMA-type & Linear serial dependence after suitable differencing or transformation \\
Exponential smoothing & Level/trend/seasonality dominate and a strong statistical baseline is needed \\
State-space & Latent dynamics, missing observations, or explicit state uncertainty matter \\
Tree-based ML & Many engineered covariates and nonlinear interactions are available \\
RNN/LSTM & Sequential nonlinear structure is supported by sufficient data \\
Transformer-based & Multi-series/covariate settings justify the additional model capacity and compute \\
\bottomrule
\end{tabular}
\end{table}

\begin{block}{Evaluation rule}
Compare against simple baselines under the same forecast origins, horizons, and loss functions. Architectural complexity is useful only if it improves the actual forecasting objective.
\end{block}
\end{frame}

\begin{frame}{Summary \& Best Practices}
\textbf{Key takeaways for time series analysis}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Core Principles:}

\begin{enumerate}
\item \textbf{Temporal ordering matters}
   \begin{itemize}
   \item Preserve the information set that would be available at each forecast origin
   \item Use time-based train/test splits
   \end{itemize}

\item \textbf{Check stationarity}
   \begin{itemize}
   \item Diagnose trend, seasonality, breaks, and stochastic persistence
   \item Use unit-root/stationarity tests as evidence within a broader specification analysis
   \end{itemize}

\item \textbf{Understand your data}
   \begin{itemize}
   \item Decompose into components
   \item Identify seasonality, trend
   \item Check for outliers, missing values
   \end{itemize}

\item \textbf{Start simple, add complexity}
   \begin{itemize}
   \item Baseline: naive, moving average
   \item Then statistical or structured regression/state-space models
   \item Add ML/DL only when covariates, data volume, nonlinear structure, or cross-series learning justify the complexity
   \end{itemize}

\item \textbf{Validate properly}
   \begin{itemize}
   \item Walk-forward validation
   \item Multiple forecast horizons
   \item Business-relevant metrics
   \end{itemize}
\end{enumerate}
\end{column}

\begin{column}{0.5\textwidth}
\textbf{Common Pitfalls:}

\begin{itemize}
\item[$\times$] Using random train/test split
\item[$\times$] Ignoring non-stationarity
\item[$\times$] Data leakage from future
\item[$\times$] Over-differencing
\item[$\times$] Ignoring domain knowledge
\item[$\times$] Not handling seasonality
\item[$\times$] Using only RMSE (check MAE, MAPE too)
\end{itemize}
