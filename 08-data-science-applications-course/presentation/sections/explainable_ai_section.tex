\begin{frame}{Why Interpretability Matters}
\textbf{Accuracy and interpretability are not a single universal trade-off.}

\begin{itemize}
  \item Simple models are often easier to inspect directly.
  \item Complex models can be harder to inspect, but may or may not improve task performance.
  \item Compare interpretable baselines and complex models on the actual task.
  \item Use post-hoc explanations only for questions they can validly answer.
  \item Evaluate explanation fidelity and stability rather than assuming transparency.
\end{itemize}

\begin{alertblock}{Methodological point}
Interpretability is a property of the model, explanation method, audience, and question being asked; it is not a single scalar that trades off mechanically against accuracy.
\end{alertblock}
\end{frame}

\begin{frame}{Why Explanations Are Used}
\begin{columns}
\begin{column}{0.48\textwidth}
\textbf{Model development}
\begin{itemize}
  \item debugging and error analysis;
  \item detecting leakage or unintended shortcuts;
  \item communicating model behavior to domain experts;
  \item generating hypotheses for further analysis.
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Governance and decisions}
\begin{itemize}
  \item documentation and auditability;
  \item contestability and human oversight;
  \item analysis of disparities and failure modes;
  \item communication with affected stakeholders.
\end{itemize}
\end{column}
\end{columns}

\begin{alertblock}{Methodological caution}
Fairness audits can reveal disparities under specific statistical definitions, but explanation methods alone do not establish causality, fairness, or the appropriate normative standard for a decision system.
\end{alertblock}
\end{frame}

\begin{frame}{Interpretability: Three Useful Axes}
\begin{columns}
\begin{column}{0.48\textwidth}
\textbf{Intrinsic vs. post-hoc}
\begin{itemize}
  \item \textbf{Intrinsic}: the model form itself supports direct inspection.
  \item \textbf{Post-hoc}: an additional method is applied after fitting to summarize behavior.
\end{itemize}

\textbf{Model-specific vs. model-agnostic}
\begin{itemize}
  \item \textbf{Model-specific}: uses internal structure or gradients.
  \item \textbf{Model-agnostic}: queries predictions without requiring model internals.
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Global vs. local}
\begin{itemize}
  \item \textbf{Global}: summarizes behavior across a population or region of feature space.
  \item \textbf{Local}: explains one prediction or a small neighborhood.
\end{itemize}

\textbf{Question first}
\begin{itemize}
  \item What behavior is being explained?
  \item For which population or instance?
  \item To which audience?
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Interpretability Toolbox}
\begin{table}
\small
\begin{tabular}{lll}
\toprule
\textbf{Method} & \textbf{Typical scope} & \textbf{Access} \\
\midrule

\begin{block}{Levels of analysis}
Algorithm transparency, global model behavior, and local prediction explanations answer different questions. One method rarely covers all three well.
\end{block}
\end{frame}

\begin{frame}{Intrinsically Interpretable Models: Regression}
\textbf{Linear regression}
\[
  \hat{y}=\beta_0+\beta_1x_1+\cdots+\beta_px_p.
\]

\begin{itemize}
  \item Coefficients summarize model-conditional associations on the response scale.
  \item Sign and magnitude depend on units, coding, interactions, and the other variables included.
  \item A coefficient is not automatically a causal effect.
\end{itemize}

\textbf{Logistic regression}
\[
  \log\left(\frac{p}{1-p}\right)=\beta_0+\beta_1x_1+\cdots.
\]

\begin{itemize}
  \item $\exp(\beta_j)$ is an odds ratio under the specified model.
  \item Probability-scale effects vary with the covariate pattern and baseline risk.
\end{itemize}
\end{frame}

\begin{frame}{SHAP: Shapley-Value Attribution}
SHAP methods define feature attributions using Shapley values from cooperative game theory.

\[
\phi_i = \sum_{S\subseteq N\setminus\{i\}}
\frac{|S|!(|N|-|S|-1)!}{|N|!}
\left[v(S\cup\{i\})-v(S)\right].
\]

\begin{itemize}
  \item the value function $v(S)$ specifies what it means for a subset of features to be present;
  \item attributions depend on the chosen background/reference distribution;
  \item local additivity holds relative to the SHAP formulation and baseline.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{SHAP in Practice and Caveats}
\begin{lstlisting}[style=code,basicstyle=\ttfamily\small]
explainer = shap.TreeExplainer(model)
values = explainer(X_test)

shap.plots.waterfall(values[0])
shap.plots.beeswarm(values)
\end{lstlisting}

\begin{itemize}
  \item local plots decompose one prediction relative to a baseline;
  \item aggregated SHAP summaries describe the fitted model, not the data-generating mechanism;
  \item correlated features can make attribution depend strongly on the conditional/interventional semantics used by the explainer;
  \item SHAP values are not causal effects.
\end{itemize}

\begin{alertblock}{Interpretation}
The axioms characterize an attribution scheme once the value function is defined. They do not guarantee that the explanation is stable, causal, or appropriate for every audience.
\end{alertblock}
\end{frame}

\begin{frame}[fragile,allowframebreaks]{LIME: Local Interpretable Model-Agnostic Explanations}
\textbf{Explain any model locally with simple approximations}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{The LIME Approach:}

\textbf{Key Idea:} Approximate complex model locally with interpretable model

\vspace{0.2cm}

\textbf{Algorithm:}
\begin{enumerate}
\item Select instance to explain
\item Generate perturbed samples nearby
\item Get predictions from black box
\item Train simple model (linear/tree) on perturbed data
\item Use simple model to explain
\end{enumerate}

\vspace{0.3cm}

\textbf{Mathematical Formulation:}

\[
\xi(x) = \argmin_{g \in \mathcal{G}} \mathcal{L}(f, g, \pi_x) + \Omega(g)
\]

where:
\begin{itemize}
\item $f$: Black box model
\item $g$: Simple interpretable model
\item $\pi_x$: Proximity measure
\item $\Omega(g)$: Complexity penalty
\end{itemize}
\end{column}

\begin{column}{0.5\textwidth}
\textbf{LIME Implementation:}

\begin{verbatim}
from lime import lime_tabular

# Create explainer
explainer = lime_tabular.
    LimeTabularExplainer(
        X_train,
        feature_names=feature_names,
        class_names=['No', 'Yes'],
        mode='classification'
    )

# Explain instance
explanation = explainer.explain_instance(
    X_test[0],
    model.predict_proba,
    num_features=10
)

explanation.show_in_notebook()
\end{verbatim}

\vspace{0.2cm}

\end{table}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Global Importance: Permutation and Drop-Column}
\textbf{Permutation importance}
\begin{itemize}
  \item permute one feature and measure the change in a chosen validation metric;
  \item model-agnostic, but sensitive to feature dependence and the evaluation distribution;
  \item can underestimate importance when correlated predictors substitute for one another.
\end{itemize}

\textbf{Drop-column importance}
\begin{itemize}
  \item refit the model without a feature and compare validation performance;
  \item measures the value of including that feature under the full retraining procedure;
  \item more computationally expensive and still model/procedure dependent.
\end{itemize}
\end{frame}

\begin{frame}{Feature Effects: PDP, ICE, and Tree Importance}
\begin{itemize}
  \item \textbf{PDP}: averages predictions over the empirical distribution of other features.
  \item \textbf{ICE}: retains one curve per observation to expose heterogeneous fitted responses.
  \item \textbf{Tree impurity importance}: fast and model-specific, but can be biased toward variables with many possible split points.
\end{itemize}

\begin{alertblock}{Use methods by question}
Importance, marginal effect plots, and local attributions answer different questions. Combining several methods can be useful, but it does not create a single complete or causal explanation.
\end{alertblock}
\end{frame}

\begin{frame}[fragile,allowframebreaks]{Counterfactual Explanations}
\textbf{"What would need to change for a different outcome?"}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{The Counterfactual Question:}

"Why was my loan denied?"

\textbf{Bad Answer:}
"Credit score was too low (SHAP value = -0.3)"

\textbf{Better Answer (Counterfactual):}
"If your credit score were 680 instead of 620, you would be approved"

\vspace{0.3cm}

\textbf{Properties of Good Counterfactuals:}

\begin{enumerate}
\item \textcolor{forest}{\textbf{Proximity:}} Close to original instance
\item \textcolor{navyblue}{\textbf{Sparsity:}} Change few features
\item \textcolor{crimson}{\textbf{Actionability:}} Can actually change
\item \textcolor{purple}{\textbf{Plausibility:}} Realistic values
\end{enumerate}

\vspace{0.2cm}

\textbf{Mathematical Formulation:}

\[
\argmin_{x'} d(x, x') + \lambda \cdot \mathbb{1}[f(x') \neq y']
\]

Find nearest instance with different prediction.
\end{column}

\begin{column}{0.5\textwidth}
\textbf{Methods:}

\begin{enumerate}
\item \textbf{Optimization-Based:}
   \begin{itemize}
   \item Minimize distance subject to prediction constraint
   \item Can use genetic algorithms
   \end{itemize}

\item \textbf{DiCE} (Diverse Counterfactual Explanations):
   \begin{itemize}
   \item Generate multiple diverse counterfactuals
   \item Gives user options
   \end{itemize}
\end{enumerate}

\vspace{0.2cm}

\textbf{Implementation:}

\begin{verbatim}
import dice_ml

# Create DiCE explainer
dice = dice_ml.Dice(data, model)

# Generate counterfactuals
cf = dice.generate_counterfactuals(
    instance,
    total_CFs=3,
    desired_class="opposite"
)
cf.visualize_as_dataframe()
\end{verbatim}

\begin{frame}[fragile,allowframebreaks]{Knowledge Check: Explainable AI}
\textbf{Test your understanding}

\vspace{0.2cm}

\textbf{Question 1:} What is the key difference between SHAP and LIME?

\begin{enumerate}[A)]
\item SHAP is for trees only, LIME is model-agnostic
\item \textcolor{forest}{\textbf{SHAP has theoretical guarantees (Shapley values), LIME uses local approximation}}
\item LIME is faster
\item They are the same
\end{enumerate}

\pause

\begin{block}{Explanation}
\textcolor{forest}{\textbf{Answer: B}} - SHAP methods are tied to Shapley-value axioms under a specified value function/background distribution, while LIME fits a local surrogate under a chosen perturbation and weighting scheme. Neither label alone guarantees faithful, stable, or causal explanations.
\end{block}

\pause

\vspace{0.2cm}

\textbf{Question 2:} When would you use counterfactual explanations over feature importance?

\begin{enumerate}[A)]
\item When you need global model understanding
\item \textcolor{forest}{\textbf{When users need actionable advice on how to change outcomes}}
\item When the model is a decision tree
\item Never, feature importance is always better
\end{enumerate}

\pause

\begin{block}{Explanation}
\textcolor{forest}{\textbf{Answer: B}} - Counterfactuals answer "what would I need to change?" which is actionable. Example: "Increase income by \$10K to get approved" vs. "Income has high feature importance" (not actionable). Perfect for loan applications, college admissions, medical interventions where people want to know how to achieve different outcomes.
\end{block}
\end{frame}

\begin{frame}[fragile,allowframebreaks]{Summary \& Best Practices}
\textbf{Building interpretable ML systems}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{The Interpretability Workflow:}

\begin{enumerate}
\item \textbf{Model Selection:}
   \begin{itemize}
   \item Start with interpretable baselines
   \item Document accuracy-interpretability tradeoff
   \item Justify complexity if needed
   \end{itemize}

\item \textbf{Global Understanding:}
   \begin{itemize}
   \item Permutation importance
   \item Partial dependence plots
   \item SHAP summary plots
   \end{itemize}

\item \textbf{Local Explanations:}
   \begin{itemize}
   \item SHAP force plots for key predictions
   \item LIME for individual cases
   \item Counterfactuals for recourse
   \end{itemize}

\item \textbf{Validation:}
   \begin{itemize}
   \item Check explanations make sense
   \item Verify with domain experts
   \item Test on edge cases
   \end{itemize}

\item \textbf{Communication:}
   \begin{itemize}
   \item Tailor to audience
   \item Visualize clearly
   \item Provide actionable insights
   \end{itemize}
\end{enumerate}
\end{column}
