\section{Motivation and Examples}
%==================================================

\begin{frame}{Why Time Series?}
  \begin{itemize}
    \item Many real-world phenomena are naturally ordered in time:
      \begin{itemize}
        \item Financial returns, volatility, prices.
        \item Temperature and climate indicators.
        \item Sensor data, biomedical signals, internet traffic.
      \end{itemize}
    \item We often observe \textbf{one long history}, not repeated experiments.
    \item Aim:
      \begin{itemize}
        \item Understand the probabilistic mechanism generating the series.
        \item Forecast future values.
        \item Detect changes, regimes, or anomalies.
      \end{itemize}
  \end{itemize}

  \vspace{0.3cm}
  Stationarity describes invariance under time shifts; ergodic or mixing conditions connect one observed path to long-run population quantities.
\end{frame}

\begin{frame}{Typical Time Series Questions}
  \begin{itemize}
    \item Is the mean level of the series stable over time?
    \item Does the variability change (volatility clustering, heteroscedasticity)?
    \item Are there trends or seasonal patterns?
    \item Can we treat one observed trajectory as \emph{representative} of the process?
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Stationarity} addresses stability of distributions over time.  
  \textbf{Ergodicity} connects time averages and ensemble averages.
\end{frame}

%==================================================
\section{Stochastic Processes and Perspectives}
%==================================================

\begin{frame}{Stochastic Process: Interaction with Time}
  \begin{itemize}
    \item A time series is a realization of a stochastic process:
      \[
        \{X_t\}_{t \in \mathbb{Z}} \quad \text{or} \quad \{X_t\}_{t \in \mathbb{R}}.
      \]
    \item For each time $t$, $X_t$ is a random variable defined on some probability space.
    \item A realization (sample path) is one function $t \mapsto x_t$ generated by the process.
    \item In practice we observe:
      \[
        x_1, x_2, \dots, x_T.
      \]
  \end{itemize}

  The challenge: infer properties of the \emph{process} from a single time-ordered sample.
\end{frame}

\begin{frame}{Two Perspectives: Time and Ensemble}
  \begin{itemize}
    \item \textbf{Time perspective} (within one realization):
      \begin{itemize}
        \item Follow $X_t$ for $t=1,\dots,T$.
        \item Compute time averages, such as sample mean and sample variance.
      \end{itemize}
    \item \textbf{Ensemble perspective} (across realizations):
      \begin{itemize}
        \item At fixed time $t$, consider many copies $X_t^{(1)}, \dots, X_t^{(N)}$.
        \item The distribution of these is the \emph{ensemble distribution} at time $t$.
      \end{itemize}
    \item In practice, we almost never have many independent realizations from the same process.
  \end{itemize}

  \vspace{0.3cm}
  This tension between time and ensemble is where stationarity and ergodicity enter.
\end{frame}

\begin{frame}{Fundamental Problem}
  \begin{itemize}
    \item We want ensemble quantities, such as $E[X_t]$, $\mathrm{Var}(X_t)$, or joint distributions.
    \item We only have a single path:
      \[
        x_1, x_2, \dots, x_T.
      \]
    \item Natural estimators:
      \[
        \bar{X}_T = \frac{1}{T} \sum_{t=1}^T X_t,
        \quad
        s_T^2 = \frac{1}{T-1} \sum_{t=1}^T (X_t - \bar{X}_T)^2.
      \]
    \item When is $\bar{X}_T$ a good estimator of $E[X_t]$?  
          When does $s_T^2$ estimate $\mathrm{Var}(X_t)$?
  \end{itemize}

  \vspace{0.2cm}
  Answering this rigorously requires assumptions tailored to the statistic of interest: stationarity may define the target moments, while ergodic, mixing, or other law-of-large-numbers conditions justify estimating them from one long path.
\end{frame}

%==================================================
\section{Stationarity}
%==================================================

\begin{frame}{Strict (Strong) Stationarity}
  \textbf{Definition.}  
  A process $\{X_t\}$ is \emph{strictly stationary} if for any $k \in \mathbb{N}$, any times
  \[
    t_1, \dots, t_k
  \]
  and any integer shift $h$, the joint distributions satisfy
  \[
    (X_{t_1}, \dots, X_{t_k}) \overset{d}{=} (X_{t_1+h}, \dots, X_{t_k+h}).
  \]

  \vspace{0.3cm}
  Intuition:
  \begin{itemize}
    \item The probabilistic structure of the process is invariant under time shifts.
    \item All finite-dimensional distributions are time-homogeneous.
  \end{itemize}
\end{frame}

\begin{frame}{Weak (Covariance) Stationarity}
  \textbf{Definition.}  
  A process $\{X_t\}$ is \emph{weakly stationary} (or covariance stationary) if:
  \begin{enumerate}
    \item $E[X_t] = \mu$ is constant for all $t$.
    \item $\mathrm{Var}(X_t) = \sigma^2 < \infty$ is constant for all $t$.
    \item $\mathrm{Cov}(X_t, X_{t+h})$ depends only on $h$, not on $t$:
      \[
        \gamma(h) = \mathrm{Cov}(X_t, X_{t+h}).
      \]
  \end{enumerate}

  \vspace{0.3cm}
  This is enough for many linear time series models (ARMA, etc.) and for spectral analysis.
\end{frame}

\begin{frame}{Autocovariance and Autocorrelation}
  For a weakly stationary process:
  \begin{itemize}
    \item Autocovariance function (ACVF):
      \[
        \gamma(h) = \mathrm{Cov}(X_t, X_{t+h}), \quad h \in \mathbb{Z}.
      \]
    \item Autocorrelation function (ACF):
      \[
        \rho(h) = \frac{\gamma(h)}{\gamma(0)}.
      \]
  \end{itemize}

  \vspace{0.3cm}
  Properties:
  \begin{itemize}
    \item $\gamma(0) = \sigma^2$.
    \item $\gamma(-h) = \gamma(h)$.
    \item $\rho(0) = 1$, and $|\rho(h)| \le 1$ for all $h$.
  \end{itemize}
\end{frame}

\begin{frame}{Examples of Stationary Processes}
  \textbf{Example 1: White noise}
  \begin{itemize}
    \item $X_t \sim \text{i.i.d.}(0, \sigma^2)$.
    \item $E[X_t] = 0$, $\gamma(0) = \sigma^2$, $\gamma(h) = 0$ for $h \neq 0$.
    \item Strictly and weakly stationary.
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Example 2: AR(1) with $| \phi | < 1$}
  \begin{align*}
    X_t &= \phi X_{t-1} + \varepsilon_t, \\
    \varepsilon_t &\sim \text{i.i.d.}(0, \sigma_\varepsilon^2), \quad |\phi|<1.
  \end{align*}
  Then:
  \[
    E[X_t] = 0, \quad
    \gamma(h) = \frac{\sigma_\varepsilon^2}{1 - \phi^2} \phi^{|h|}.
  \]
  This process is weakly stationary and, under mild conditions, strictly stationary.
\end{frame}

\begin{frame}{Non-stationary Example: Random Walk}
  \textbf{Random walk:}
  \begin{align*}
    X_t &= X_{t-1} + \varepsilon_t, \\
    \varepsilon_t &\sim \text{i.i.d.}(0, \sigma_\varepsilon^2),
    \quad X_0 \text{ given}.
  \end{align*}

  \vspace{0.2cm}
  \begin{itemize}
    \item $E[X_t] = E[X_0]$ (constant mean).
    \item $\mathrm{Var}(X_t) = \mathrm{Var}(X_0) + t \sigma_\varepsilon^2$ grows with $t$.
    \item The variance is not constant \textrightarrow{} not even weakly stationary.
  \end{itemize}

  \vspace{0.2cm}
  Differencing $\Delta X_t = X_t - X_{t-1} = \varepsilon_t$ recovers a stationary series (white noise).
\end{frame}

\begin{frame}{Strict vs Weak Stationarity}
  \begin{itemize}
    \item Strict stationarity $\Rightarrow$ weak stationarity if second moments exist.
    \item Weak stationarity does not necessarily imply strict stationarity.
    \item In practice:
      \begin{itemize}
        \item We rarely can test strict stationarity.
        \item Most modelling frameworks assume weak stationarity.
      \end{itemize}
  \end{itemize}

  \vspace{0.3cm}
  For many Gaussian processes, weak stationarity actually implies strict stationarity, because the distribution is fully determined by mean and covariance.
\end{frame}

\section{Ergodicity}
%==================================================

\begin{frame}{Time Averages vs Ensemble Averages}
  For a weakly stationary process $\{X_t\}$ with finite mean $\mu$:
  \begin{itemize}
    \item Ensemble mean:
      \[
        \mu = E[X_t].
      \]
    \item Time average (sample mean) over one trajectory:
      \[
        \bar{X}_T = \frac{1}{T}\sum_{t=1}^T X_t.
      \]
  \end{itemize}

  \vspace{0.3cm}
  \textbf{Question:} Under what conditions does
  \[
    \bar{X}_T \xrightarrow[T \to \infty]{\text{(some sense)}} \mu?
  \]
  This is the core idea behind \emph{ergodicity}.
\end{frame}

\begin{frame}{Ergodicity in the Mean}
  \textbf{Definition (informal).}  
  A stationary process $\{X_t\}$ is \emph{ergodic in the mean} if
  \[
    \bar{X}_T = \frac{1}{T}\sum_{t=1}^T X_t
    \xrightarrow[T \to \infty]{\text{a.s. or in prob.}}
    E[X_t] = \mu.
  \]

  \vspace{0.3cm}
  Interpretation:
  \begin{itemize}
    \item A single long realization is enough to estimate the mean.
    \item The time average along one trajectory converges to the ensemble expectation.
  \end{itemize}
\end{frame}

\begin{frame}{Ergodicity for Higher Moments}
  We can extend the notion of ergodicity:
  \begin{itemize}
    \item \textbf{Ergodic in variance:}
      \[
        \frac{1}{T}\sum_{t=1}^T (X_t - \bar{X}_T)^2
        \xrightarrow[T \to \infty]{}
        \mathrm{Var}(X_t).
      \]
    \item \textbf{Ergodic for autocovariances:}
      \[
        \frac{1}{T}\sum_{t=1}^{T-h} X_t X_{t+h}
        \xrightarrow[T \to \infty]{}
        \gamma(h).
      \]
  \end{itemize}

  \vspace{0.3cm}
  In practice, we estimate $\gamma(h)$ from one series and implicitly assume ergodicity.
\end{frame}

\begin{frame}{Ergodic Theorem (Very Informal)}
  \textbf{Birkhoff's Ergodic Theorem (informal statement):}
  \begin{itemize}
    \item Consider a measure-preserving transformation $T$ on a probability space and an integrable function $f$.
    \item Under ergodicity of $T$, the time average along orbits converges (almost surely) to the space average:
      \[
        \frac{1}{n}\sum_{k=0}^{n-1} f(T^k \omega)
        \xrightarrow[n\to\infty]{\text{a.s.}}
        \int f \, d\mathbb{P}.
      \]
  \end{itemize}

  \vspace{0.3cm}
  Time averages are legitimate estimators of expectations if the underlying dynamical system is ergodic.
\end{frame}

\begin{frame}{Stationarity and Ergodicity Are Different Assumptions}
  \begin{itemize}
    \item \textbf{Stationarity} concerns invariance of the process distribution, or of selected moments, under time shifts.
    \item \textbf{Ergodicity} concerns whether time averages of suitable functions converge to population averages along a stationary process.
    \item In time-series theory, ergodicity is normally formulated for a stationary process (or stationary measure-preserving shift). It is therefore an additional property, not a replacement for stationarity.
    \item A stationary process can be non-ergodic: for example, a mixture of stationary regimes in which each realization remains permanently in one latent regime.
  \end{itemize}
\end{frame}
\begin{frame}{Examples and Conditions for Ergodicity}
  \begin{itemize}
    \item Causal ARMA processes driven by i.i.d. innovations are stationary and ergodic under the usual stable-root conditions.
    \item For stationary Gaussian processes, ergodicity is not automatic; it is characterized by the absence of atoms in the spectral measure.
    \item Stronger assumptions such as mixing are often used when one needs central limit theorems, standard-error formulas, or consistency of more complicated statistics.
    \item A stationary mixture of noncommunicating regimes provides a standard example of stationarity without ergodicity.
  \end{itemize}
  \begin{alertblock}{Statistical implication}
    Mean ergodicity can justify a sample average, but consistency of variances, autocovariances, quantiles, or nonlinear estimators may require additional moment and dependence conditions.
  \end{alertblock}
\end{frame}

\section{Practical Diagnostics}

%==================================================

\begin{frame}{Diagnosing Temporal Stability in Practice}
  \begin{itemize}
    \item \textbf{Visual and structural checks}: inspect trend, seasonality, breaks, changing variance, and changes in dependence rather than reducing the question to one test statistic.
    \item \textbf{ADF / DF-type tests}: test a unit-root null within a chosen autoregressive specification; conclusions depend on deterministic terms, lag selection, sample size, and breaks.
    \item \textbf{KPSS}: tests level- or trend-stationarity as the null, again under a specified deterministic structure.
    \item \textbf{ACF/PACF}: useful descriptive diagnostics for persistence and candidate linear models, but a slowly decaying ACF is not a proof of a unit root and a sharp-looking cutoff is not a general stationarity test.
  \end{itemize}
  \begin{alertblock}{Do not treat tests as an oracle}
    Near-unit-root processes, structural breaks, seasonal roots, and short samples can make apparently conflicting test results entirely plausible.
  \end{alertblock}
\end{frame}
\begin{frame}{Modeling Nonstationarity}
  \begin{itemize}
    \item \textbf{Deterministic trend}: model or remove a deterministic component when substantively justified.
    \item \textbf{Stochastic trend / unit root}: differencing can be appropriate, but overdifferencing can create unnecessary moving-average structure and inflate forecast variance.
    \item \textbf{Seasonality}: use deterministic seasonal terms, seasonal differencing, or explicit seasonal state components according to the mechanism.
    \item \textbf{Changing scale}: log or Box--Cox transformations may stabilize scale-dependent variance, but they also change the interpretation of errors and forecasts.
    \item \textbf{Structural breaks or regime changes}: transforming the series is not a substitute for modeling the break or allowing parameters to change.
  \end{itemize}
\end{frame}
\begin{frame}{Ergodicity in Applied Modeling}
  \begin{itemize}
    \item Ergodicity is usually justified through the assumed stochastic model rather than by a standalone empirical test.
    \item If the fitted model is stationary and ergodic and the relevant moments exist, time averages can consistently estimate corresponding population expectations.
    \item Consistency of sample autocovariances or other nonlinear summaries may require additional dependence and moment conditions.
    \item Structural breaks, persistent regime changes, and model misspecification can invalidate the theoretical argument even when a short sample looks visually stable.
  \end{itemize}
\end{frame}

\section{Summary and Key Messages}

\begin{frame}{Synthesis}
  \begin{itemize}
    \item Strict and weak stationarity are invariance assumptions; ergodicity is a separate property connecting a single long trajectory to population quantities.
    \item Stationarity alone does not justify treating time averages as population expectations, and mean ergodicity alone does not automatically justify every higher-order estimator.
    \item Unit-root and stationarity tests are conditional on a specification and can have low power or be distorted by structural breaks and near-unit-root behavior.
    \item Differencing, detrending, seasonal adjustment, and variance-stabilizing transforms solve different modeling problems and should not be used interchangeably.
    \item The relevant assumption is the one required by the model and estimand: forecast validity, parameter inference, spectral analysis, and long-run averages need different forms of temporal stability.
  \end{itemize}
\end{frame}

\section{Appendix: Technical Notes}

\begin{frame}{Spectral View (Optional)}
  A weakly stationary process has a spectral measure. When the autocovariance sequence is absolutely summable, that measure has a continuous density
  \[
    f(\lambda) = \frac{1}{2\pi} \sum_{h=-\infty}^{\infty} \gamma(h) e^{-i \lambda h},
    \quad \lambda \in [-\pi, \pi].
  \]

  \vspace{0.3cm}
  \begin{itemize}
    \item The spectrum describes how second-order variation is distributed over frequencies.
    \item Causal ARMA processes satisfy the regularity needed for the familiar rational spectral density.
  \end{itemize}
\end{frame}

\begin{frame}{Conditions for Stationary AR(1)}
  Consider
  \[
    X_t = \phi X_{t-1} + \varepsilon_t, \quad \varepsilon_t \sim \text{i.i.d.}(0, \sigma_\varepsilon^2).
  \]

  \vspace{0.3cm}
  \textbf{Key facts:}
  \begin{itemize}
    \item If $|\phi| < 1$, there exists the standard causal strictly stationary solution
      \[
        X_t = \sum_{j=0}^{\infty} \phi^j \varepsilon_{t-j}.
      \]
    \item For that causal solution:
      \[
        E[X_t] = 0, \quad
        \mathrm{Var}(X_t) = \frac{\sigma_\varepsilon^2}{1 - \phi^2}.
      \]
    \item At $|\phi|=1$, the usual finite-variance stationary solution fails. For $|\phi|>1$, a noncausal stationary solution can be constructed using future innovations, but it is not the forecasting representation normally intended by an AR model.
  \end{itemize}
\end{frame}
